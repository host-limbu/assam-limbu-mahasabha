/* ============================================================ ADMIN LOGIN — Assam Limbu Mahasabha Authentication, role validation, and redirect ============================================================ */ (function() { 'use strict'; // DOM references const form = document.getElementById('login-form'); const emailInput = document.getElementById('email'); const passwordInput = document.getElementById('password'); const loginBtn = document.getElementById('login-btn'); const errorMessage = document.getElementById('error-message'); const loadingSpinner = document.getElementById('loading-spinner'); // ============================================================ // LOGIN HANDLER // ============================================================ if (form) { form.addEventListener('submit', function(e) { e.preventDefault(); // Clear previous error hideError(); // Validate inputs const email = emailInput.value.trim(); const password = passwordInput.value.trim(); if (!email || !password) { showError('Please enter both email and password.'); return; } // Show loading state setLoading(true); // Attempt sign in firebase.auth().signInWithEmailAndPassword(email, password) .then(function(userCredential) { const user = userCredential.user; // Check if user has a valid role const role = AdminCommon.getUserRole(user.uid); if (!role) { // User is authenticated but not authorized firebase.auth().signOut(); showError('Access denied. You do not have admin privileges.'); setLoading(false); return; } // Store role in session (optional) sessionStorage.setItem('adminRole', role); sessionStorage.setItem('adminUID', user.uid); // Redirect to dashboard window.location.href = 'dashboard.html'; }) .catch(function(error) { setLoading(false); var errorMsg = error.message; // User-friendly error messages if (error.code === 'auth/user-not-found') { errorMsg = 'No account found with this email address.'; } else if (error.code === 'auth/wrong-password') { errorMsg = 'Incorrect password. Please try again.'; } else if (error.code === 'auth/invalid-email') { errorMsg = 'Please enter a valid email address.'; } else if (error.code === 'auth/too-many-requests') { errorMsg = 'Too many failed attempts. Please try again later.'; } showError(errorMsg); }); }); } // ============================================================ // ENTER KEY SUPPORT (already handled by form submit) // ============================================================ // ============================================================ // UI HELPERS // ============================================================ function showError(msg) { errorMessage.textContent = msg; errorMessage.classList.add('visible'); } function hideError() { errorMessage.classList.remove('visible'); } function setLoading(loading) { if (loading) { loadingSpinner.classList.add('visible'); loginBtn.disabled = true; loginBtn.textContent = 'Signing in...'; } else { loadingSpinner.classList.remove('visible'); loginBtn.disabled = false; loginBtn.textContent = 'Sign In'; } } // ============================================================ // CHECK IF ALREADY LOGGED IN // ============================================================ firebase.auth().onAuthStateChanged(function(user) { if (user) { // Check if user has a valid role const role = AdminCommon.getUserRole(user.uid); if (role) { // Already logged in, redirect to dashboard sessionStorage.setItem('adminRole', role); sessionStorage.setItem('adminUID', user.uid); window.location.href = 'dashboard.html'; } else { // User logged in but no role - sign them out firebase.auth().signOut(); } } }); console.log('login.js loaded successfully.'); })();
